cmake_minimum_required(VERSION 3.15)
project(FxStudioCounter C CXX)
include(GenerateG1A)
find_package(Gint 2.9 REQUIRED)
add_executable(counter src/main.cpp)
target_compile_features(counter PRIVATE cxx_std_17)
target_compile_options(counter PRIVATE -Wall -Wextra)
target_link_libraries(counter stdc++ Gint::Gint)
generate_g1a(TARGET counter OUTPUT "Counter.g1a" NAME "Counter" INTERNAL "@FXCOUNT" ICON "assets/icon.png")
