cmake_minimum_required(VERSION 3.15)
project(FileMapProbe C ASM)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
add_executable(probe3 src/start.S src/main.c src/resident.c src/resident_syscalls.S src/callback.S)
target_compile_options(probe3 PRIVATE -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra)
target_link_options(probe3 PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=probe3.map)
set_property(TARGET probe3 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET probe3 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/relocations.py" fixups "$<TARGET_FILE:probe3>")
generate_g1a(TARGET probe3 OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Probe3.g1a" NAME "Probe3" INTERNAL "@PROBE3" ICON "assets/icon.png")
add_custom_command(TARGET probe3 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/bundle.py" "${CMAKE_CURRENT_BINARY_DIR}")
