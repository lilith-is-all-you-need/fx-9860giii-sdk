cmake_minimum_required(VERSION 3.15)
project(LoaderProbe C ASM)
include(GenerateG1A)
if(NOT Python3_EXECUTABLE)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
set(flags -m4-nofpu -Os -ffreestanding -fno-builtin -fno-stack-protector -fno-unwind-tables -fno-asynchronous-unwind-tables -ffunction-sections -fdata-sections -Wall -Wextra)
add_executable(probe2 src/start.S src/main.c src/resident.c src/resident_syscalls.S src/callback.S)
target_compile_options(probe2 PRIVATE ${flags})
target_link_options(probe2 PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld" -Wl,--emit-relocs,-Map=probe2.map)
set_property(TARGET probe2 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/probe.ld")
add_custom_command(TARGET probe2 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/pack.py" fixups "$<TARGET_FILE:probe2>")
generate_g1a(TARGET probe2 OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Probe2.g1a" NAME "Probe2" INTERNAL "@PROBE2" ICON "assets/icon.png")
add_executable(payload src/payload.S src/payload.c src/payload_syscalls.S)
target_compile_options(payload PRIVATE ${flags})
target_link_options(payload PRIVATE -m4-nofpu -T "${CMAKE_CURRENT_SOURCE_DIR}/src/payload.ld")
set_property(TARGET payload APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/payload.ld")
generate_g1a(TARGET payload OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/P2Target.container" NAME "P2Target" INTERNAL "@P2TEST" ICON "assets/icon.png")
add_dependencies(probe2 payload)
add_custom_command(TARGET probe2 POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/pack.py" bundle "${CMAKE_CURRENT_BINARY_DIR}")
