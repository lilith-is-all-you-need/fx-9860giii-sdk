cmake_minimum_required(VERSION 3.15)
project(FxStudioGraphicsAudit C)
include(GenerateG1A)
include(Fxconv)
find_package(Gint 2.9 REQUIRED)
set(ASSETS assets/sprite.png assets/digits.png)
fxconv_declare_assets(${ASSETS} WITH_METADATA)
add_executable(graphics_audit main.c ${ASSETS})
target_link_libraries(graphics_audit Gint::Gint)
generate_g1a(TARGET graphics_audit OUTPUT "GraphicsAudit.g1a" NAME "GRAPHICS" INTERNAL "@FXGRAPH")
