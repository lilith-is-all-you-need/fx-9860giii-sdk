cmake_minimum_required(VERSION 3.24)
project(FxStudioWin32 LANGUAGES CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio x64")
endif()
set(EDITOR_DEPS "" CACHE PATH "Directory prepared by editor_deps.py")
if(NOT EXISTS "${EDITOR_DEPS}/Scintilla.h")
  message(FATAL_ERROR "Run editor_deps.py and set EDITOR_DEPS")
endif()
add_executable(FxStudioWin32 WIN32 main.cpp platform.cpp workspace.cpp views.cpp commands.cpp app.rc)
set_target_properties(FxStudioWin32 PROPERTIES OUTPUT_NAME "FxStudio.Win32" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_features(FxStudioWin32 PRIVATE cxx_std_17)
target_compile_definitions(FxStudioWin32 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(FxStudioWin32 PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/permissive-;/MP;/EHsc>")
target_precompile_headers(FxStudioWin32 PRIVATE <platform.h>)
target_include_directories(FxStudioWin32 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${EDITOR_DEPS}")
target_link_libraries(FxStudioWin32 PRIVATE comctl32 comdlg32 shell32 shlwapi ole32 uuid gdiplus dwmapi uxtheme bcrypt windowscodecs)
target_link_options(FxStudioWin32 PRIVATE /MANIFEST:NO)
